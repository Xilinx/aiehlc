# tiny_llama in aiednn eager mode

LLaMA-style decoder (TinyLlama's shape, scaled down): RMSNorm, RoPE, GQA (8 query / 2 KV heads), SwiGLU; prefill on [1, S, 512]; input `[1, 128, 512]` (random embeddings; random weights). Quantized to int8 and lowered through the aiednn MLIR pipeline, it becomes **49 aiednn primitives** in the generated C app. The eager plan of the primary build (`tiny_llama_int8_mlir_8x4_eager`, mesh 8x4, baseline) makes **5 kernel instances** that serve **45 of those layers**; on the board `aiednn_stats` counts **433 AIE launches** in the timed run. Each instance is one shape-specialized template instantiation, loaded right before the layer it serves runs.

## Run it

```bash
python example/aiednn/eager/models/run.py --model tiny_llama                # x86 emulation, 4x4
python example/aiednn/eager/models/run.py --model tiny_llama --show         # + plan and instance code
python example/aiednn/eager/models/run.py --model tiny_llama --mesh 8x4 --target vek280 --board galerina13 \
    --vitis-settings <.../Vitis/settings64.sh>
# the same through the model runner (builds worklocal/aiednn/models/tiny_llama_int8_mlir_8x4[_warm]_eager):
python example/aiednn/models/run_models.py --models tiny_llama --modes int8 --app vek280 --board galerina13 \
    --eager --eager-mesh 8x4 --mlir [--tag warm]
```

`--tag warm` only names the build directory; the warm launch is in the current code.

## 1. The PyTorch model

From [`example/aiednn/models/zoo.py`](../../../../example/aiednn/models/zoo.py) (`RMSNorm`, `rope`, `LlamaBlock`, `TinyLlama`):

```python
class RMSNorm(nn.Module):
    def __init__(self, d, eps=1e-6):
        super().__init__()
        self.eps, self.weight = eps, nn.Parameter(torch.ones(d))

    def forward(self, x):
        return x * torch.rsqrt(x.pow(2).mean(-1, keepdim=True) + self.eps) * self.weight


def rope(t, cos, sin):
    """rotary embedding, the half-split (LLaMA / HF) form"""
    h = t.shape[-1] // 2
    return t * cos + torch.cat([-t[..., h:], t[..., :h]], dim=-1) * sin


class LlamaBlock(nn.Module):
    def __init__(self, d, heads, kv_heads, ff):
        super().__init__()
        self.h, self.kvh, self.hd = heads, kv_heads, d // heads
        self.n1, self.n2 = RMSNorm(d), RMSNorm(d)
        self.q = nn.Linear(d, d, bias=False)
        self.k = nn.Linear(d, kv_heads * self.hd, bias=False)
        self.v = nn.Linear(d, kv_heads * self.hd, bias=False)
        self.o = nn.Linear(d, d, bias=False)
        self.gate, self.up = nn.Linear(d, ff, bias=False), nn.Linear(d, ff, bias=False)
        self.down = nn.Linear(ff, d, bias=False)

    def forward(self, x, cos, sin):
        b, s, d = x.shape
        h = self.n1(x)
        q = rope(self.q(h).reshape(b, s, self.h, self.hd).permute(0, 2, 1, 3), cos, sin)
        k = rope(self.k(h).reshape(b, s, self.kvh, self.hd).permute(0, 2, 1, 3), cos, sin)
        v = self.v(h).reshape(b, s, self.kvh, self.hd).permute(0, 2, 1, 3)
        rep = self.h // self.kvh                        # GQA: each K / V head serves rep query heads
        k, v = k.repeat_interleave(rep, dim=1), v.repeat_interleave(rep, dim=1)
        a = F.scaled_dot_product_attention(q, k, v, is_causal=True)
        x = x + self.o(a.permute(0, 2, 1, 3).reshape(b, s, d))
        h = self.n2(x)
        return x + self.down(F.silu(self.gate(h)) * self.up(h))   # SwiGLU gated MLP


class TinyLlama(nn.Module):
    """LLaMA-style decoder (TinyLlama's shape, scaled down): RMSNorm, RoPE, GQA (8 query / 2 KV heads),
    SwiGLU; prefill on [1, S, 512]"""

    def __init__(self, layers=4, d=512, heads=8, kv_heads=2, ff=1408, seq=128):
        super().__init__()
        self.blocks = nn.ModuleList([LlamaBlock(d, heads, kv_heads, ff) for _ in range(layers)])
        self.norm = RMSNorm(d)
        hd = d // heads
        inv = 1.0 / (10000 ** (torch.arange(0, hd, 2).float() / hd))
        f = torch.outer(torch.arange(seq).float(), inv)
        emb = torch.cat([f, f], dim=-1)
        self.register_buffer("cos", emb.cos()[None, None])
        self.register_buffer("sin", emb.sin()[None, None])

    def forward(self, x):
        for b in self.blocks:
            x = b(x, self.cos, self.sin)
        return self.norm(x)
```

Model and input shape, from `zoo.BENCH`: `"tiny_llama": (TinyLlama, (1, 128, 512))`

// M a multiple of HW_ROWS * TM, N of 4 (mesh column pn owns columns pn * N / 4 .. as one block: every port's block
// walk stays two deep, which the kernel analysis can route), K of TK, TK of 4 (whole DMA words)
template <int M, int N, int K, int TK = 64, int TM = 16>
__global__ void gemm_s8(aie::port<input_window_int8 *, aie::IFM, aie::shape<M, K>> a,
                        aie::port<input_window_int8 *, aie::WEIGHTS, aie::shape<N, K>> w,
